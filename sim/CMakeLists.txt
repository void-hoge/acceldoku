find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(verilator REQUIRED HINTS $ENV{VERILATOR_ROOT})
find_program(IVERILOG_EXECUTABLE NAMES iverilog REQUIRED)
find_program(VVP_EXECUTABLE NAMES vvp REQUIRED)
find_program(YOSYS_EXECUTABLE NAMES yosys REQUIRED)
include("${PROJECT_SOURCE_DIR}/cmake/RtlTests.cmake")
set(ACCELDOKU_DATASET "${CMAKE_CURRENT_SOURCE_DIR}/data/17puz49158.txt" CACHE FILEPATH "Test dataset")
set(ACCELDOKU_BASELINE "" CACHE FILEPATH "Full core-test CSV baseline for regression")
foreach(suite IN ITEMS deduction_engine branch_generator acceldoku_core acceldoku acceldoku_axi async_queue reference)
    add_subdirectory(${suite})
endforeach()

foreach(top IN ITEMS acceldoku_core acceldoku acceldoku_axi acceldoku_kv260)
    set(suite "${top}")
    if(top STREQUAL "acceldoku_core")
        set(sources ${ACCELDOKU_CORE_RTL})
        set(params)
    elseif(top STREQUAL "acceldoku")
        set(sources ${ACCELDOKU_STREAM_RTL})
        set(params -GCORES=${ACCELDOKU_CORES})
    else()
        set(sources ${ACCELDOKU_AXI_RTL})
        set(params -GCORES=${ACCELDOKU_CORES})
        if(top STREQUAL "acceldoku_kv260")
            # Board wrapper checks belong with the AXI integration boundary.
            set(suite acceldoku_axi)
            list(APPEND sources "${PROJECT_SOURCE_DIR}/fpga/kv260/rtl/acceldoku_kv260.sv")
        endif()
    endif()
    add_test(NAME lint-${top} COMMAND "${VERILATOR_BIN}" --lint-only --language 1800-2017
        -Wall --top-module ${top} ${params} ${sources})
    set_tests_properties(lint-${top} PROPERTIES LABELS "${suite}" TIMEOUT 120)
endforeach()
add_custom_target(lint COMMAND "${CMAKE_CTEST_COMMAND}" --output-on-failure -R "^lint-"
    WORKING_DIRECTORY "${PROJECT_BINARY_DIR}" USES_TERMINAL VERBATIM)
add_custom_target(formal COMMAND "${CMAKE_CTEST_COMMAND}" --output-on-failure -R "^formal-"
    WORKING_DIRECTORY "${PROJECT_BINARY_DIR}" USES_TERMINAL VERBATIM)
